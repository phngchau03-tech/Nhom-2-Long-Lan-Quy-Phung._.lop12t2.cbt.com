    <script>
        // Chuyển đổi giữa các tab
        function switchTab(tabName) {
            const membersSection = document.getElementById('tab-members');
            const assignmentsSection = document.getElementById('tab-assignments');
            const btnMembers = document.getElementById('btn-members');
            const btnAssignments = document.getElementById('btn-assignments');

            if (tabName === 'members') {
                membersSection.classList.remove('hidden');
                assignmentsSection.classList.add('hidden');
                btnMembers.classList.add('bg-indigo-700');
                btnAssignments.classList.remove('bg-indigo-700');
            } else {
                assignmentsSection.classList.remove('hidden');
                membersSection.classList.add('hidden');
                btnAssignments.classList.add('bg-indigo-700');
                btnMembers.classList.remove('bg-indigo-700');
            }
        }

        // Quản lý dữ liệu bài tập trong LocalStorage
        let assignments = JSON.parse(localStorage.getItem('assignments')) || [];

        function saveToLocalStorage() {
            localStorage.setItem('assignments', JSON.stringify(assignments));
        }

        // Render danh sách bài tập ra giao diện
        function renderAssignments() {
            const listContainer = document.getElementById('assignments-list');
            listContainer.innerHTML = '';

            if (assignments.length === 0) {
                listContainer.innerHTML = '<p class="text-gray-500 col-span-2 text-center">Chưa có bài tập nào được đăng tải.</p>';
                return;
            }

            assignments.forEach((item, index) => {
                const card = document.createElement('div');
                card.className = 'bg-white p-6 rounded-xl shadow-md flex flex-col justify-between space-y-4';

                card.innerHTML = `
                    <div>
                        <div class="flex justify-between items-start">
                            <h4 class="text-xl font-bold text-gray-800">${item.title}</h4>
                            <span class="text-xs bg-indigo-100 text-indigo-800 px-2.5 py-1 rounded-full font-medium">Bởi: ${item.author}</span>
                        </div>
                        <p class="text-gray-600 mt-2 text-sm">${item.description}</p>
                    </div>
                    ${item.image ? `<div class="mt-2"><img src="${item.image}" alt="Bài tập" class="w-full h-48 object-cover rounded-md border"></div>` : ''}
                    <div class="flex justify-end space-x-2 pt-2 border-t">
                        <button onclick="editAssignment(${index})" class="px-3 py-1 bg-amber-500 text-white text-sm rounded hover:bg-amber-600 transition">Chỉnh sửa</button>
                        <button onclick="deleteAssignment(${index})" class="px-3 py-1 bg-red-500 text-white text-sm rounded hover:bg-red-600 transition">Xóa</button>
                    </div>
                `;
                listContainer.appendChild(card);
            });
        }

        // Xử lý Thêm / Cập nhật bài tập
        function handleFormSubmit(event) {
            event.preventDefault();

            const title = document.getElementById('title').value;
            const author = document.getElementById('author').value;
            const description = document.getElementById('description').value;
            const imageInput = document.getElementById('image-input');
            const editIndex = document.getElementById('edit-index').value;

            const processData = (imageUrl) => {
                const newItem = {
                    title,
                    author,
                    description,
                    image: imageUrl || (editIndex !== '' ? assignments[editIndex].image : '')
                };

                if (editIndex === '') {
                    assignments.push(newItem);
                } else {
                    assignments[editIndex] = newItem;
                }

                saveToLocalStorage();
                renderAssignments();
                resetForm();
            };

            if (imageInput.files && imageInput.files[0]) {
                const reader = new FileReader();
                reader.onload = function (e) {
                    processData(e.target.result);
                };
                reader.readAsDataURL(imageInput.files[0]);
            } else {
                processData(null);
            }
        }

        // Đưa dữ liệu vào form để chỉnh sửa
        function editAssignment(index) {
            const item = assignments[index];
            document.getElementById('title').value = item.title;
            document.getElementById('author').value = item.author;
            document.getElementById('description').value = item.description;
            document.getElementById('edit-index').value = index;

            document.getElementById('form-title').innerText = 'Chỉnh Sửa Bài Tập';
            document.getElementById('submit-btn').innerText = 'Cập Nhật Bài Tập';
            document.getElementById('cancel-edit-btn').classList.remove('hidden');

            window.scrollTo({ top: 400, behavior: 'smooth' });
        }

        // Xóa bài tập
        function deleteAssignment(index) {
            if (confirm('Bạn có chắc chắn muốn xóa bài tập này không?')) {
                assignments.splice(index, 1);
                saveToLocalStorage();
                renderAssignments();
            }
        }

        // Reset form về trạng thái thêm mới
        function resetForm() {
            document.getElementById('assignment-form').reset();
            document.getElementById('edit-index').value = '';
            document.getElementById('form-title').innerText = 'Thêm Bài Tập Mới';
            document.getElementById('submit-btn').innerText = 'Đăng Bài Tập';
            document.getElementById('cancel-edit-btn').classList.add('hidden');
        }

        // Khởi chạy khi tải trang
        renderAssignments();
    </script>
